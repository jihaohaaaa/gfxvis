  "      return;",

  "          break;",
  "        }",
  "        curr = curr.left;",
  "      } else {",
  "        if (!curr.right) {",
  "          curr.right = node;",
  "          break;",
  "        }",
  "        curr = curr.right;",
  "      }",
  "    }",
  "  }",
  "",
  "  inOrder(node: TreeNode<T> | null = this.root, res: T[] = []): T[] {",

  "    this.inOrder(node.right, res);",
  "    return res;",
  "  }",

## 五、架构安全性与超时防护设计

将代码执行接入操作系统本地原生环境带来了强大的交互力，但同时也伴随着<strong>计算安全与防死循环需求</strong>。在底层设计中，本执行器贯彻了以下防护原则：

1. <strong>唯一随机沙箱隔离</strong>：每次请求均由 `randomUUID()` 生成专用的临时文件目录，执行结束无论成败立即递归擦除，杜绝磁盘文件残留；
2. <strong>8 秒超时强杀熔断</strong>：如果读者在编辑代码时不小心写了 `while(true)` 或死递归，子进程监视器将在 <strong>8000 毫秒</strong> 后自动发送 `SIGTERM` 强杀子进程并向前端输出拦截报告，确保不会占满或阻塞服务器资源；
3. <strong>环境探针自适应</strong>：在执行 C++ 或 Rust 前，Action 会首先进行环境变量与命令行探针检查；若宿主机未预装编译环境，将友好提供包管理器安装指令，而非抛出粗糙的崩溃信息。
